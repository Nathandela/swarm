package skeleton

import (
	"os"
	"path/filepath"
	"testing"
)

func TestOverlaySourceConfigDriftRefusesSuccessorBeforeKill(t *testing.T) {
	x := feasibleAccountTest(t, 2)
	if err := x.m.reportFailure(x.m.w, x.source.ID, "quota", "", "native-failure"); err != nil {
		t.Fatal(err)
	}
	x.m.step()
	if got := x.m.w.state.AccountRotations[x.source.ID].State; got != accountReserved {
		t.Fatalf("first step state=%q, want %q", got, accountReserved)
	}
	if err := os.WriteFile(filepath.Join(x.original, "config.toml"), []byte("model_reasoning_effort=\"medium\"\n"), 0600); err != nil {
		t.Fatal(err)
	}
	x.m.step()
	rec := x.m.w.state.AccountRotations[x.source.ID]
	if rec.State != accountReserved || rec.LastError != "successor-preflight-refused" || len(x.f.killed) != 0 || len(x.f.launched) != 0 {
		t.Fatalf("drift result state=%q last_error=%q kills=%d launches=%d", rec.State, rec.LastError, len(x.f.killed), len(x.f.launched))
	}
}
