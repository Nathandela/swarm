package discord

import (
	"context"
	"fmt"
	"github.com/bwmarrin/discordgo"
	"strings"
	"testing"
	"time"
)

func TestTeammatePlanSpike(t *testing.T) {
	s, workers, messenger := serviceRig(t)
	client, _ := NewDiscordClient("fixture")
	b := NewBridge(s, client)
	target := workers.rows[0]
	workers.rows = nil
	event := &discordgo.MessageCreate{Message: &discordgo.Message{ID: "41", GuildID: "10", ChannelID: "20", Content: "<@100> thoughts?", Timestamp: time.Now(), Author: &discordgo.User{ID: "2", Username: "Alice"}, Mentions: []*discordgo.User{{ID: "100"}}}}
	b.Message(context.Background(), event)
	if len(workers.sends) != 0 || len(messenger.posts) != 1 || !strings.Contains(messenger.posts[0].Text, "no agents") {
		t.Fatal("unexpected empty roster behavior", workers.sends, messenger.posts)
	}
	t.Log("empty roster: 0 native requests; human received the empty directory notice")
	workers.rows = []Worker{target}
	if err := s.Join(Route{GuildID: "10", ChannelID: "20"}, target.Target); err != nil {
		t.Fatal(err)
	}
	event.Mentions = nil
	event.Author = &discordgo.User{ID: "3", Username: "Bob"}
	for i := 0; i < 110; i++ {
		event.ID = fmt.Sprint(1000 + i)
		event.Content = fmt.Sprint("follow up ", i)
		b.Message(context.Background(), event)
	}
	if len(workers.sends) != 0 || len(s.recent["20"]) != 50 || s.recent["20"][0].ID != "1060" {
		t.Fatal("unexpected plain follow-up or context handling", len(workers.sends), s.recent["20"])
	}
	t.Log("110 untagged human follow-ups on a joined route: 0 native requests; 50 messages retained; 60 evicted")
	route, err := (ScopedMessenger{Config: s.Config}).Resolve(context.Background(), Route{GuildID: "10", ChannelID: "20"})
	if err != nil || route.ParentID != "" {
		t.Fatal(route, err)
	}
	t.Log("guild-only scope accepts unclassified channel without fetching parent or verifying its thread type")
}
