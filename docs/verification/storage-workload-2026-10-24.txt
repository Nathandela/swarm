		})
	}
}

func TestManifestRejectsSymlinkAndNonregularFiles(t *testing.T) {
	for _, kind := range []string{"symlink", "hardlink", "fifo", "oversized"} {
		t.Run(kind, func(t *testing.T) {
			root := t.TempDir()
			path := filepath.Join(root, executableManifest)
			switch kind {
			case "symlink", "hardlink":
				target := filepath.Join(t.TempDir(), "empty")
				if err := os.WriteFile(target, nil, 0600); err != nil {
					t.Fatal(err)
				}
				var err error
				if kind == "symlink" {
					err = os.Symlink(target, path)
				} else {
					err = os.Link(target, path)
				}
				if err != nil {
					t.Fatal(err)
				}
			case "fifo":
				if err := syscall.Mkfifo(path, 0600); err != nil {
					t.Fatal(err)
				}
			case "oversized":
				f, err := os.Create(path)
				if err != nil {
					t.Fatal(err)
				}
				if err = f.Truncate(1<<20 + 1); err != nil {
					_ = f.Close()
					t.Fatal(err)
				}
				if err = f.Close(); err != nil {
					t.Fatal(err)
				}
			}
			done := make(chan error, 1)
			go func() { _, err := registeredExecutables(root); done <- err }()
			select {
			case err := <-done:
				if err == nil {
					t.Fatalf("accepted unsafe %s manifest", kind)
				}
			case <-time.After(time.Second):
				if kind == "fifo" {
					fd, err := syscall.Open(path, syscall.O_WRONLY|syscall.O_NONBLOCK, 0)
					if err == nil {
						_ = syscall.Close(fd)
					}
				}
				t.Fatalf("blocked on %s manifest", kind)
			}
		})
	}
}

func TestSupervisorParentTerminationStillCleans(t *testing.T) {
	reportPath := filepath.Join(t.TempDir(), "report")
	exe, _ := os.Executable()
	cmd := exec.Command(exe, "-test.run=^TestSupervisorFixture$", "-test.timeout=30s")
	for _, env := range os.Environ() {
		if !strings.HasPrefix(env, runRootEnv+"=") {
			cmd.Env = append(cmd.Env, env)
		}
	}
	cmd.Env = append(cmd.Env, "TESTBIN_SCENARIO=timeout", "TESTBIN_REPORT="+reportPath)
	if err := cmd.Start(); err != nil {
		t.Fatal(err)
	}
	exited := make(chan error, 1)
	waitDone := make(chan struct{})
	go func() { defer close(waitDone); exited <- cmd.Wait() }()
	t.Cleanup(func() {
		_ = cmd.Process.Kill()
		select {
		case <-waitDone:
		case <-time.After(5 * time.Second):
		}
	})
	var report survivorReport
	deadline := time.Now().Add(5 * time.Second)
	for {
		raw, err := os.ReadFile(reportPath)
		if err == nil && json.Unmarshal(raw, &report) == nil {
			break
		}
		if time.Now().After(deadline) {
			t.Fatal("child did not report readiness")
		}
		time.Sleep(10 * time.Millisecond)
	}
	identity := processcontain.Identity{PID: report.PID, StartTime: report.Start}
	t.Cleanup(func() {
		if identity.Alive() {
			_ = syscall.Kill(report.PID, syscall.SIGKILL)
		}
		_ = os.RemoveAll(report.Root)
	})
	if err := cmd.Process.Signal(syscall.SIGTERM); err != nil {
		t.Fatal(err)
	}
	select {
	case err := <-exited:
		if err == nil {
			t.Fatal("lost terminated child status")
		}
	case <-time.After(10 * time.Second):
		t.Fatal("supervisor did not finish")
	}
	if identity.Alive() {
		t.Error("parent TERM left owned detached survivor")
	}
	if _, err := os.Stat(report.Root); !os.IsNotExist(err) {
		t.Errorf("parent TERM retained root: %v", err)
	}
}

func workloadScratchMeasure(root string) (int64, int64, error) {
	var allocated, logical int64
	seen := map[[2]uint64]bool{}
	err := filepath.Walk(root, func(path string, info os.FileInfo, err error) error {
		if os.IsNotExist(err) {
			return nil
		}
		if err != nil {
			return err
		}
		stat, ok := info.Sys().(*syscall.Stat_t)
		if !ok {
			return fmt.Errorf("missing allocation metadata")
		}
		key := [2]uint64{uint64(stat.Dev), uint64(stat.Ino)}
		if seen[key] {
			return nil
		}
		seen[key] = true
		allocated += stat.Blocks * 512
		if info.Mode().IsRegular() {
			logical += info.Size()
		}
		return nil
	})
	return allocated, logical, err
}
func TestWorkloadAbortedScratchAllocated(t *testing.T) {
	for cycle := 1; cycle <= 3; cycle++ {
		for _, mode := range []string{"panic", "timeout"} {
			reportPath := filepath.Join(t.TempDir(), "report")
			self, err := os.Executable()
			if err != nil {
				t.Fatal(err)
			}
			cmd := exec.Command(self, "-test.run=^TestSupervisorFixture$", "-test.timeout=500ms")
			for _, env := range os.Environ() {
				if !strings.HasPrefix(env, runRootEnv+"=") && !strings.HasPrefix(env, "TESTBIN_SCENARIO=") {
					cmd.Env = append(cmd.Env, env)
				}
			}
			cmd.Env = append(cmd.Env, "TESTBIN_SCENARIO="+mode, "TESTBIN_REPORT="+reportPath, "WORKLOAD_SAMPLE=1")
			output, err := cmd.CombinedOutput()
			exit, ok := err.(*exec.ExitError)
			if !ok || exit.ExitCode() != 2 {
				t.Fatalf("child %s status %v", mode, err)
			}
			raw, err := os.ReadFile(reportPath)
			if err != nil {
				t.Fatalf("report: %v\n%s", err, output)
			}
			var report survivorReport
			if err := json.Unmarshal(raw, &report); err != nil {
				t.Fatal(err)
			}
			identity := processcontain.Identity{PID: report.PID, StartTime: report.Start}
			t.Cleanup(func() {
				if identity.Alive() {
					_ = syscall.Kill(report.PID, syscall.SIGKILL)
				}
				_ = os.RemoveAll(report.Root)
			})
			var peakAllocated, peakLogical int64
			found := false
			for _, line := range strings.Split(string(output), "\n") {
				if strings.HasPrefix(line, "WORKLOAD_SCRATCH_READY ") {
					if _, err := fmt.Sscanf(line, "WORKLOAD_SCRATCH_READY allocated=%d logical=%d", &peakAllocated, &peakLogical); err != nil {
						t.Fatal(err)
					}
					found = true
				}
			}
			if !found || peakAllocated == 0 {
				t.Fatalf("missing real scratch sample: %s", output)
			}
			if identity.Alive() {
				t.Fatal("owned survivor remained alive")
			}
			if _, err := os.Stat(report.Root); !os.IsNotExist(err) {
				t.Fatal("aborted scratch root remained")
			}
			t.Logf("WORKLOAD name=aborted_scratch mode=%s cycle=%d peak_allocated=%d peak_logical=%d settled_allocated=0 settled_logical=0 samples=2 cadence=child_ready_and_parent_exit", mode, cycle, peakAllocated, peakLogical)
		}
	}
}
